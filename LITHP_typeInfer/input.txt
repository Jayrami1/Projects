;;; -------------------------------------------
;;; 10. Functions & Recursion
;;; -------------------------------------------
; Anonymous function evaluation
((lambda (x y) (+ x y)) 100 200)
; Function definition (Defun)
(defun square (x) (* x x))
((lambda (f x) (f x)) square 5)
; Recursive function using label (Factorial)
(label fact (lambda (n) (cond ((= n 0) 1) (t (* n (fact (- n 1)))))))

;;; 1. Define a function 'square' intended for Integers
(defun square (x) (* x x))

;;; 2. Apply it correctly (Int -> Int)
(square 5)

;;; 3. Apply it with a mismatched type (Bool -> Int)
(square ())

;;; 11. INTENTIONAL ERRORS (Watch these fail gracefully!)
;;; -------------------------------------------

; Arithmetic Error: Adding Int and Bool

; Comparison Error: Comparing Bool and Int

; List Error 1: Getting car of an empty list

; List Error 2: Getting cdr of an empty list
(cdr ())

; List Error 3: Consing to a non-list

; Cond Error: Condition is an Int, not a Bool

; Logic Error: using 'and' on an Int
